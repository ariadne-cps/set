cmake_minimum_required(VERSION 3.15)
project(AriadneSetTutorial LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}")
find_package(AriadneSet REQUIRED)
add_executable(tutorial_set tutorial_set.cpp)
target_link_libraries(tutorial_set PRIVATE AriadneSet::ariadne-set)
